1. Armstrong Number

An Armstrong number equals the sum of its digits, each raised to the power of the number of digits.

Sample input: 153

function checkArmstrong(number) {
    const digits = String(number).split("");
    const power = digits.length;
    const sum = digits.reduce((total, digit) =>
        total + Number(digit) ** power, 0);

    console.log("Number:", number);
    console.log("Sum:", sum);
    console.log(sum === number
        ? "Armstrong number"
        : "Not an Armstrong number");
}
checkArmstrong(153);

2. Even or Odd

A number divisible by 2 with remainder 0 is even; otherwise it is odd.

Sample input: 24

function checkEvenOrOdd(number) {
    if (number % 2 === 0) {
        console.log(number + " is Even");
    } else {
        console.log(number + " is Odd");
    }
}
checkEvenOrOdd(24);

3. Fibonacci Series

Each term is the sum of the two previous terms. This series starts with 0 and 1.

Sample input: 10 terms

function fibonacciSeries(terms) {
    let first = 0, second = 1;
    let series = [];

    for (let i = 0; i < terms; i++) {
        series.push(first);
        let next = first + second;
        first = second;
        second = next;
    }

    console.log("Fibonacci Series:", series.join(", "));
}
fibonacciSeries(10);

4. Perfect Number

A perfect number equals the sum of its positive divisors excluding itself. Example: 6 = 1 + 2 + 3.

Sample input: 28

function checkPerfectNumber(number) {
    let sum = 0;

    for (let i = 1; i <= number / 2; i++) {
        if (number % i === 0) {
            sum += i;
        }
    }

    console.log("Sum of proper divisors:", sum);
    console.log(sum === number
        ? "Perfect number"
        : "Not a Perfect number");
}
checkPerfectNumber(28);

5. Prime Number

A prime number is greater than 1 and has exactly two factors: 1 and itself.

Sample input: 17

function checkPrime(number) {
    if (number <= 1) {
        console.log("Not a Prime number");
        return;
    }

    let isPrime = true;
    for (let i = 2; i <= Math.sqrt(number); i++) {
        if (number % i === 0) {
            isPrime = false;
            break;
        }
    }

    console.log(isPrime
        ? number + " is Prime"
        : number + " is Not Prime");
}
checkPrime(17);
